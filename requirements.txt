# Minimal dependency set for the qBraid environment.
#
# Qiskit's transpiler changes between minor releases, and this task scores
# candidates by counting gates the transpiler emits. Different Qiskit versions
# therefore give different absolute scores. That is handled -- the evaluator
# fingerprints its cached baselines with the Qiskit version and recomputes when
# it changes, so scores are always internally consistent -- but pin the minor
# version in the published environment so numbers are comparable between users.
qiskit>=2.4,<2.6

# The evolutionary search engine.
shinka-evolve>=0.0.7

# Config files.
pyyaml>=6.0

# qBraid compute lifecycle, for the cross-machine path in qbraid_remote_gpu.py.
# Present on a Lab instance's system Python, but not inside a qBraid env, so the
# environment needs it explicitly.
qbraid-core>=0.6.5

# The decoder task (task_decoder/): Stim samples the surface-code circuits,
# PyMatching is the MWPM baseline, BeliefMatching is the reference decoder the
# search is asked to reach.
stim>=1.16.0
pymatching>=2.4.0
beliefmatching>=0.2.0

# Notebooks. Drop these two if the environment is CLI-only.
jupyterlab>=4.0
matplotlib>=3.8
